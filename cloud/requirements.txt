# --- Framework web ---
fastapi
uvicorn[standard]        # 1 worker apenas, conforme decisão do Blueprint
python-multipart         # necessario para os campos Form()/UploadFile de /diagnostico

# --- Inferência ---
# O carregador aceita dois runtimes, nesta ordem (app/core/inference.py):
#   1. tflite-runtime: o do Blueprint, com wheel para Linux x86_64 (Railway)
#      e aarch64, mas NAO para Windows.
#   2. ai-edge-litert: sucessor oficial do mesmo projeto, com wheel para
#      Windows, Linux e aarch64. E o que permite rodar a API localmente
#      no Windows; sem ele, a inferencia caia no placeholder em silencio.
# Nenhum dos dois e o pacote `tensorflow` completo, que continua proibido.
# O ai-edge-litert fica ATIVO porque e ele que sobe no Railway: tem wheel
# para Linux x86_64 em toda versao recente de Python, enquanto o
# tflite-runtime parou no 3.11 e deixaria o deploy refem da versao que o
# builder escolher. Sem nenhum dos dois instalado, a API sobe, responde
# 200 e devolve "indisponivel" em toda captura, sem erro visivel.
ai-edge-litert
# tflite-runtime  # alternativa; o carregador aceita os dois
numpy

# --- Banco de dados (Supabase Postgres) ---
sqlalchemy[asyncio]
asyncpg
alembic                   # migrations

# --- Storage (Supabase Storage, S3-compatible) ---
supabase                  # cliente oficial (cobre Postgres + Storage)

# --- Processamento de imagem (100% em memória, via BytesIO) ---
pillow

# --- Autenticação do dispositivo (API Key) e do dashboard (JWT/bcrypt) ---
# bcrypt usado diretamente (nao via passlib): passlib esta sem manutencao
# e tem incompatibilidade conhecida com versoes recentes do pacote bcrypt.
pyjwt
bcrypt

# --- Configuração ---
python-dotenv
pydantic-settings
