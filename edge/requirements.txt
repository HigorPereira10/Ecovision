# Dependências do software do dispositivo (Raspberry Pi OS Lite, 64 bits).
# Instaladas por edge/scripts/install.sh. O Picamera2 vem do apt
# (python3-picamera2), porque depende das bibliotecas do libcamera do
# sistema; por isso o ambiente virtual usa --system-site-packages.

pillow                       # recompressão JPEG; requer libjpeg-turbo no sistema
RPi.GPIO                     # acionamento do LED
Adafruit-Blinka              # módulo `board`, usado pelo DHT22
adafruit-circuitpython-dht   # DHT22; requer libgpiod3 no sistema
smbus2                       # ADS1115 por acesso direto ao barramento I2C
requests
python-dotenv                # carrega edge/.env nas execuções do cron
